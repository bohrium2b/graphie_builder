#import "@preview/cetz:0.2.0": canvas, plot
// Oscillatory signal synthesis engine
let amp = 1.0
let omega = 1.0
let phi = 0.0
#canvas({
/* Config Cartesian grid & axes */
plot.plot(size: (8, 4), x-tick-step: calc.pi / 2, {
plot.add-grid(stroke: 0.4pt + #e2e7f0)
// Primary Sine Carrier Wave
plot.add(
domain: (-calc.pi, 2 * calc.pi),
style: (stroke: 1.8pt + #0b57d0),
label: $"sin"(theta)$,
t => amp * calc.sin(omega * t + phi)
)
// Quadrature Cosine Reference
plot.add(
domain: (-calc.pi, 2 * calc.pi),
style: (stroke: (paint: #4b2faa, dash: "dashed")),
label: $"cos"(theta)$,
t => amp * calc.cos(omega * t)
)
})
})
